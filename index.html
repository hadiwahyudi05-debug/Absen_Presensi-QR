Kode Google Apps Script (Auto Sync 2 Arah)

Salin kode Apps Script berikut ke Google Sheets Anda. Mendukung pemrosesan Rombel, Siswa, Log Presensi, dan Akun Wali Kelas secara otomatis.


const SECRET_TOKEN = "SEC_SD_DIGITAL_2026";

function doGet(e) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  
  var logsSheet = ss.getSheetByName("Logs");
  var logsData = logsSheet ? logsSheet.getDataRange().getValues() : [];
  
  var rombelSheet = ss.getSheetByName("Rombel");
  var rombelData = rombelSheet ? rombelSheet.getDataRange().getValues() : [];
  
  var siswaSheet = ss.getSheetByName("Siswa");
  var siswaData = siswaSheet ? siswaSheet.getDataRange().getValues() : [];

  var usersSheet = ss.getSheetByName("Users");
  var usersData = usersSheet ? usersSheet.getDataRange().getValues() : [];

  return ContentService.createTextOutput(JSON.stringify({ 
    status: "success", 
    logs: logsData, 
    classes: rombelData, 
    students: siswaData,
    users: usersData
  })).setMimeType(ContentService.MimeType.JSON);
}

function doPost(e) {
  try {
    var payload = JSON.parse(e.postData.contents);
    if (!payload.token || payload.token !== SECRET_TOKEN) {
      return ContentService.createTextOutput(JSON.stringify({ status: "unauthorized", message: "Akses Ditolak" })).setMimeType(ContentService.MimeType.JSON);
    }
    var ss = SpreadsheetApp.getActiveSpreadsheet();

    if(payload.logs) {
      var logsSheet = ss.getSheetByName("Logs") || ss.insertSheet("Logs");
      logsSheet.clear();
      var headersLog = ["ID Log", "Waktu", "NIS", "Nama", "Status", "Sesi", "Telat (Menit)", "ID Kelas", "Catatan"];
      logsSheet.appendRow(headersLog);
      if(payload.logs.length > 0) {
        var logRows = payload.logs.map(log => [log.id, log.timestamp, log.nis, log.studentName || '-', log.status, log.mode, log.lateMinutes || 0, log.classId || '', log.doctorNote || log.notes || '-']);
        logsSheet.getRange(2, 1, logRows.length, headersLog.length).setValues(logRows);
      }
    }

    if(payload.classes) {
      var rombelSheet = ss.getSheetByName("Rombel") || ss.insertSheet("Rombel");
      rombelSheet.clear();
      rombelSheet.appendRow(["ID Kelas", "Nama Kelas", "Shift", "Jam Masuk", "Jam Pulang"]);
      if(payload.classes.length > 0) {
        var classRows = payload.classes.map(c => [c.id || '', c.name || '', c.shift || 'pagi', c.targetIn || '07:00', c.targetOut || '12:00']);
        rombelSheet.getRange(2, 1, classRows.length, 5).setValues(classRows);
      }
    }

    if(payload.students) {
      var siswaSheet = ss.getSheetByName("Siswa") || ss.insertSheet("Siswa");
      siswaSheet.clear();
      siswaSheet.appendRow(["NISN/NIS", "Nama Lengkap", "No WA", "ID Kelas"]);
      if(payload.students.length > 0) {
        var studentRows = payload.students.map(s => [s.nis || '', s.name || '', s.phone || '-', s.classId || '']);
        siswaSheet.getRange(2, 1, studentRows.length, 4).setValues(studentRows);
      }
    }

    if(payload.users) {
      var usersSheet = ss.getSheetByName("Users") || ss.insertSheet("Users");
      usersSheet.clear();
      usersSheet.appendRow(["ID User", "Username", "Password", "Role", "Nama", "ID Kelas"]);
      if(payload.users.length > 0) {
        var userRows = payload.users.map(u => [u.id || '', u.username || '', u.password || '', u.role || 'teacher', u.name || '', u.classId || '']);
        usersSheet.getRange(2, 1, userRows.length, 6).setValues(userRows);
      }
    }

    return ContentService.createTextOutput(JSON.stringify({ status: "success", message: "Sinkronisasi Berhasil" })).setMimeType(ContentService.MimeType.JSON);
  } catch(error) {
    return ContentService.createTextOutput(JSON.stringify({ status: "error", message: error.toString() })).setMimeType(ContentService.MimeType.JSON);
  }
}
